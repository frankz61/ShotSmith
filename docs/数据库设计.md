# ShotSmith —— 数据库设计

> PostgreSQL · 核心两张表：`task`（生成任务）1 — N `asset`（生成素材）

```sql
CREATE EXTENSION IF NOT EXISTS "pgcrypto";   -- gen_random_uuid()

-- 生成任务
CREATE TABLE task (
    id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    status        VARCHAR(16) NOT NULL DEFAULT 'pending',  -- pending/processing/success/partial/failed
    current_stage VARCHAR(16),                             -- ingest/matting/generate/postprocess/qc/package
    progress      SMALLINT    NOT NULL DEFAULT 0,          -- 0~100
    source_type   VARCHAR(8)  NOT NULL,                    -- upload / url
    source_ref    TEXT        NOT NULL,                    -- 上传文件名或图片 URL
    description   TEXT,                                    -- 可选文字描述
    options       JSONB       NOT NULL DEFAULT '{}',       -- image_types/sizes/style/candidates
    error_message TEXT,
    created_at    TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at    TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- 生成素材
CREATE TABLE asset (
    id             UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    task_id        UUID        NOT NULL REFERENCES task(id) ON DELETE CASCADE,
    type           VARCHAR(16) NOT NULL,                   -- cutout / white_bg / scene
    size_label     VARCHAR(16),                            -- 1:1 / 3:4 / 4:5
    path           TEXT        NOT NULL,
    fidelity_score NUMERIC(4,3),                           -- 还原度 0~1，低于阈值转人工复核
    qc_status      VARCHAR(16) NOT NULL DEFAULT 'pending', -- pending/passed/needs_review/rejected
    selected       BOOLEAN     NOT NULL DEFAULT false,     -- 用户是否选用
    gen_params     JSONB       NOT NULL DEFAULT '{}',      -- provider/prompt/seed
    created_at     TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX idx_task_status   ON task(status);
CREATE INDEX idx_asset_task_id ON asset(task_id);
```

> 二期再加 `batch_job`（批量）、`platform_spec`（平台尺寸规则）两张表。
